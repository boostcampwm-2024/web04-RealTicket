// 2. USER_LEVEL[session.userStatus] < USER_LEVEL[requiredStatus] → UnauthorizedException('해당 페이지에 접근할 수 없습니다.') → 401

// USER_LEVEL 상수
USER_LEVEL = {
  LOGIN: 0,
  WAITING: 1,
  ENTERING: 2,
  SELECTING_SEAT: 3,
  RECONNECTING_SELECTING: 2,
  ADMIN: 4,
}
// WAITING(1) < SELECTING_SEAT(3) → 401 (키는 존재하지만 레벨 부족)

// ABN-07: 좌석 삭제 상태 재취소 흐름
// bookSeat(sid, eventId, 0, 0, 'reserved') → 201 (점유)
// bookSeat(sid, eventId, 0, 0, 'deleted') → 201 (취소 → 좌석 상태 다시 available)
// bookSeat(sid, eventId, 0, 0, 'deleted') → 409 ('이미 취소된 좌석입니다.')
// ConflictException은 updateSeatDeleted에서: runUpdateSeatLua result=0 → 이미 available 상태

// ABN-08: bookingAmount 초과 시나리오
// setBookingCount(sid, 1) → bookingAmount=1
// bookSeat → 0번째 좌석 점유 (bookedSeats.length=1)
// bookSeat → 1번째 좌석 점유 시도 → validateAndAddBookedSeat → bookingAmount(1) <= bookedSeats.length(1) → 400

// TestRedisService.getOrThrow() → ioredis Redis 인스턴스 반환
// getRedisService(app).getOrThrow().del(key) → Redis 키 삭제

```typescript
    // ABN-07: 이미 취소된 좌석 재취소 → 409
    it('ABN-07: 이미 취소된(deleted) 좌석을 다시 취소 → 409', async () => {
      await openEventReservation(app, adminSid, eventId).expect(201);
      const userSid = await loginAsUser(app, 'abn07user1', 'pass1234');

      // SELECTING_SEAT 진입
      await requestPermission(app, userSid, eventId).expect(200);
      await setBookingCount(app, userSid, 1).expect(201);
      await transitionToSelectingSeat(app, userSid);

      // 좌석 점유 → 취소 (정상 흐름)
      await bookSeat(app, userSid, eventId, 0, 0, 'reserved').expect(201);
      await bookSeat(app, userSid, eventId, 0, 0, 'deleted').expect(201);
      // 이 시점에서 좌석 상태: available(bit=1), bookedSeats에서도 제거됨

      // 이미 취소된 좌석 재취소 → updateSeatDeleted → runUpdateSeatLua → result=0 → 409
      const res = await bookSeat(app, userSid, eventId, 0, 0, 'deleted');
      expect(res.status).toBe(409);
    });
```
